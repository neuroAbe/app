* SSH Key Setup
** Generate New SSH Key
   #+BEGIN_SRC bash
   # Navigate to SSH directory
   cd ~/.ssh

   # If directory doesn't exist, create it
   mkdir -p ~/.ssh && cd ~/.ssh

   # Generate new SSH key (replace with your email)
   ssh-keygen -t ed25519 -C "your.email@example.com"

   # When prompted:
   # "Enter file in which to save the key": Press Enter (use default)
   # "Enter passphrase": Type a secure passphrase (or press Enter for none)
   # "Enter same passphrase again": Repeat passphrase

   # Expected output:
   # Your identification has been saved in /home/username/.ssh/id_ed25519
   # Your public key has been saved in /home/username/.ssh/id_ed25519.pub
   #+END_SRC

** Verify SSH Key Created
   #+BEGIN_SRC bash
   # List SSH keys
   ls -la ~/.ssh/

   # You should see:
   # id_ed25519      (private key - NEVER share this)
   # id_ed25519.pub  (public key - this goes to GitHub/GitLab)
   #+END_SRC

** Start SSH Agent
   #+BEGIN_SRC bash
   # Start the SSH agent in background
   eval "$(ssh-agent -s)"

   # Expected output: Agent pid 12345

   # Add your SSH key to the agent
   ssh-add ~/.ssh/id_ed25519

   # If you set a passphrase, enter it when prompted
   # Expected output: Identity added: /home/username/.ssh/id_ed25519
   #+END_SRC

** Copy Public Key to Clipboard
   #+BEGIN_SRC bash
   # Option 1: Using xclip (install if needed: sudo apt install xclip)
   cat ~/.ssh/id_ed25519.pub | xclip -selection clipboard

   # Option 2: Using xsel (install if needed: sudo apt install xsel)
   cat ~/.ssh/id_ed25519.pub | xsel --clipboard

   # Option 3: Just display it and copy manually
   cat ~/.ssh/id_ed25519.pub

   # The output looks like:
   # ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIG... your.email@example.com
   #+END_SRC

* Configure Git Identity
** Set Global Git Configuration
   #+BEGIN_SRC bash
   # Set your name (use your real name)
   git config --global user.name "Your Full Name"

   # Set your email (must match GitHub account)
   git config --global user.email "your.email@example.com"

   # Set default branch name
   git config --global init.defaultBranch main

   # Set default editor (optional, choose one)
   git config --global core.editor "vim"
   # OR
   git config --global core.editor "nano"
   # OR
   git config --global core.editor "code --wait"  # VS Code

   # Enable colored output
   git config --global color.ui auto

   # Verify configuration
   git config --global --list

   # Expected output:
   # user.name=Your Full Name
   # user.email=your.email@example.com
   # init.defaultbranch=main
   # core.editor=vim
   # color.ui=auto
   #+END_SRC

** Clone Using SSH (Recommended)
   #+BEGIN_SRC bash
   # Clone the repository via SSH
   git clone git@github.com:YOUR_USERNAME/YOUR_REPO_NAME.git campaign_manager

   # Replace YOUR_USERNAME with your GitHub username
   # Replace YOUR_REPO_NAME with actual repository name

   # Example:
   # git clone git@github.com:neuroAbe/app.git campaign_manager

   # Expected output:
   # Cloning into 'campaign_manager'...
   # remote: Enumerating objects: 100, done.
   # remote: Counting objects: 100% (100/100), done.
   # remote: Compressing objects: 100% (80/80), done.
   # remote: Total 100 (delta 20), reused 100 (delta 20)
   # Receiving objects: 100% (100/100), 50.00 KiB | 1.00 MiB/s, done.
   # Resolving deltas: 100% (20/20), done.
   #+END_SRC

** Alternative: Clone Using HTTPS
   #+BEGIN_SRC bash
   # If SSH doesn't work, use HTTPS

   # Username: your GitHub username

* Troubleshooting
** SSH Key Permission Issues
   #+BEGIN_SRC bash
   # Fix SSH directory permissions
   chmod 700 ~/.ssh

   # Fix private key permissions
   chmod 600 ~/.ssh/id_ed25519

   # Fix public key permissions
   chmod 644 ~/.ssh/id_ed25519.pub

   # Fix known_hosts permissions
   chmod 644 ~/.ssh/known_hosts
   #+END_SRC

** SSH Agent Not Running
   #+BEGIN_SRC bash
   # Check if agent is running
   echo $SSH_AGENT_PID

   # If empty, start it
   eval "$(ssh-agent -s)"

   # Add key again
   ssh-add ~/.ssh/id_ed25519

   # To auto-start agent, add to ~/.bashrc or ~/.zshrc:
   echo 'eval "$(ssh-agent -s)"' >> ~/.bashrc
   echo 'ssh-add ~/.ssh/id_ed25519 2>/dev/null' >> ~/.bashrc
   #+END_SRC

** Permission Denied (publickey)
   #+BEGIN_SRC bash
   # Verify key is added to agent
   ssh-add -l

   # Should list your key
   # If empty, add it:
   ssh-add ~/.ssh/id_ed25519

   # Verify key matches GitHub
   cat ~/.ssh/id_ed25519.pub
   # Compare with key shown in GitHub Settings → SSH keys

   # Test connection with verbose output
   ssh -vT git@github.com
   # Look for "Offering public key" and "Server accepts key"
   #+END_SRC

* Complete One-Shot Setup Script
  #+BEGIN_SRC bash
  #!/bin/bash
  # Save this as setup_repo.sh and run: bash setup_repo.sh

  # Colors for output
  RED='\033[0;31m'
  GREEN='\033[0;32m'
  NC='\033[0m' # No Color

  echo -e "${GREEN}=== Campaign Manager Repository Setup ===${NC}"

  # Step 1: Check git installation
  if ! command -v git &> /dev/null; then
      echo -e "${RED}Git not installed. Installing...${NC}"
      sudo apt update && sudo apt install -y git
  fi
  echo -e "${GREEN}✓ Git installed${NC}"

  # Step 2: Configure git (replace with your info)
  echo "Configuring Git..."
  read -p "Enter your full name: " git_name
  read -p "Enter your email: " git_email
  git config --global user.name "$git_name"
  git config --global user.email "$git_email"
  echo -e "${GREEN}✓ Git configured${NC}"

  # Step 3: Generate SSH key if not exists
  if [ ! -f ~/.ssh/id_ed25519 ]; then
      echo "Generating SSH key..."
      ssh-keygen -t ed25519 -C "$git_email" -f ~/.ssh/id_ed25519 -N ""
      echo -e "${GREEN}✓ SSH key generated${NC}"
  else
      echo -e "${GREEN}✓ SSH key already exists${NC}"
  fi

  # Step 4: Start SSH agent and add key
  eval "$(ssh-agent -s)"
  ssh-add ~/.ssh/id_ed25519
  echo -e "${GREEN}✓ SSH key added to agent${NC}"

  # Step 5: Display public key
  echo ""
  echo -e "${RED}=== IMPORTANT: Copy this SSH key to GitHub ===${NC}"
  echo "Go to: GitHub → Settings → SSH Keys → New SSH Key"
  echo ""
  cat ~/.ssh/id_ed25519.pub
  echo ""
  read -p "Press Enter after you've added the key to GitHub..."

  # Step 6: Test SSH connection
  echo "Testing GitHub connection..."
  ssh -T git@github.com

  # Step 7: Clone repository
  echo ""
  read -p "Enter the SSH clone URL (git@github.com:user/repo.git): " repo_url
  mkdir -p ~/projects
  cd ~/projects
  git clone "$repo_url" campaign_manager
  cd campaign_manager

  # Step 8: Checkout feature branch
  git checkout claude/flutter-interview-showcase-01BTQPFxq2rMci5ym1ieqwcd

  echo ""
  echo -e "${GREEN}=== Setup Complete! ===${NC}"
  echo "Repository cloned to: ~/projects/campaign_manager"
  echo "Current branch: $(git branch --show-current)"
  echo ""
  echo "Next steps:"
  echo "1. cd ~/projects/campaign_manager"
  echo "2. flutter pub get"
  echo "3. flutter run"
  #+END_SRC
